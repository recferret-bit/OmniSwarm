# Key shapes shared by tools/secret-sweep.sh (git history) and infra/deploy/log-hygiene.sh (host logs).
# One rule per line: <name>|<extended regex>. Everything after the first "|" is the regex, read as-is
# by `git grep -E` and `grep -E` (no shell quoting). Every shape needs the real key's length, so this
# file, the docs that name the prefixes, and short test shapes (sk-proj-AAAA…) never match.
#   OpenRouter sk-or-v1- + 64 hex · xAI xai- + 70+ · Z.ai 32 hex "." 16 · Anthropic/OpenAI sk-ant-/sk-proj- + 40+
# Shapes that only one script needs stay in that script (PEM and NEAR keys in log-hygiene.sh, the PEM
# key body in secret-sweep.sh).
github-token|ghp_[A-Za-z0-9]{36}|github_pat_[A-Za-z0-9_]{60,}
provider-key|sk-ant-[A-Za-z0-9_-]{40,}|sk-proj-[A-Za-z0-9_-]{40,}|sk-or-v1-[0-9a-f]{64}|xai-[A-Za-z0-9]{70,}|[0-9a-f]{32}\.[A-Za-z0-9]{16}
provider-env-key|(OPENROUTER|ZAI|Z_AI|XAI|ANTHROPIC|OPENAI|GEMINI|NEARAI)_(API_KEY|AUTH_TOKEN)[[:space:]]*[=:][[:space:]]*["']?[A-Za-z0-9]{32,}
